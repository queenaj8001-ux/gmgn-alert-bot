
// ==================== v45 CLEANUP SYSTEM ====================
// Price cache for watchlist (75s TTL)
const priceCache = new Map(); // { tokenAddress: { price, timestamp, data } }
const CACHE_TTL_MS = 75 * 1000; // 75 seconds

// Cleanup configuration
const MAX_WATCHLIST_AGE_MINUTES = 90;
const MIN_VOLUME_24H_USD = 500;
const FLAT_PRICE_DURATION_MINUTES = 30;
const HIGH_RUG_SCORE_THRESHOLD = 60;
const MIN_HOLDERS_THRESHOLD = 50;
const LIQUIDITY_DROP_PERCENT = 20;
const MIN_WATCHLIST_SIZE = 50;
const GRACE_PERIOD_MINUTES = 10;
const MIN_CLEANUP_INTERVAL = 10 * 60 * 1000; // 10 minutes

let lastCleanupTime = 0;
const DRY_RUN = true; // ⚠️ Set false after dry-run verification

// Criterion 1: Token Too Old
function isTokenTooOld(token) {
  const ageMinutes = (Date.now() - token.discoveredAt) / 60000;
  if (ageMinutes < GRACE_PERIOD_MINUTES) return false;
  return ageMinutes > MAX_WATCHLIST_AGE_MINUTES;
}

// Criterion 2: Low Volume
function hasLowVolume(token) {
  return token.volume24h && token.volume24h < MIN_VOLUME_24H_USD;
}

// Criterion 3: Price Flat
function isPriceFlat(token, minutes = FLAT_PRICE_DURATION_MINUTES) {
  const history = token.recentPrices || [];
  const cutoff = Date.now() - (minutes * 60 * 1000);
  const recent = history.filter(p => p.timestamp > cutoff);
  if (recent.length < 3) return false;
  const firstPrice = recent[0].price;
  return recent.every(p => Math.abs(p.price - firstPrice) < 0.000001);
}

// Criterion 4: High Rug Risk
function hasHighRugRisk(token) {
  return token.rugCheckScore && token.rugCheckScore > HIGH_RUG_SCORE_THRESHOLD;
}

// Criterion 5: Low Holders
function hasLowHolders(token) {
  return token.holders && token.holders < MIN_HOLDERS_THRESHOLD;
}

// Criterion 6: Liquidity Drop
function hasLiquidityDrop(token, percentThreshold = LIQUIDITY_DROP_PERCENT) {
  const history = token.liquidityHistory || [];
  if (history.length < 2) return false;
  const oldLiq = history[history.length - 2].value;
  const newLiq = history[history.length - 1].value;
  if (oldLiq === 0) return false;
  const dropPercent = ((oldLiq - newLiq) / oldLiq) * 100;
  return dropPercent > percentThreshold;
}

// Quality Scoring
function calculateQualityScore(token) {
  let score = 0;
  const vol = token.volume24h || 0;
  if (vol > 50000) score += 30;
  else if (vol > 10000) score += 20;
  else if (vol > 5000) score += 15;
  else if (vol > 1000) score += 10;
  else if (vol > 500) score += 5;
  
  const liq = token.liquidity || 0;
  if (liq > 20000) score += 25;
  else if (liq > 10000) score += 20;
  else if (liq > 5000) score += 15;
  else if (liq > 2000) score += 10;
  else score += 5;
  
  const holders = token.holders || 0;
  if (holders > 500) score += 20;
  else if (holders > 200) score += 15;
  else if (holders > 100) score += 10;
  else if (holders > 50) score += 5;
  
  const mcap = token.marketCap || 0;
  if (mcap > 100000 && mcap < 10000000) score += 15;
  else if (mcap > 50000) score += 10;
  else if (mcap > 10000) score += 5;
  
  const rugScore = token.rugCheckScore || 100;
  if (rugScore < 30) score += 10;
  else if (rugScore < 50) score += 7;
  else if (rugScore < 70) score += 5;
  
  return score;
}

// Main Cleanup Function
async function cleanupWatchlist(state) {
  const now = Date.now();
  if (now - lastCleanupTime < MIN_CLEANUP_INTERVAL) {
    const timeSince = ((now - lastCleanupTime) / 1000).toFixed(0);
    console.log(`⏭️ Cleanup too soon, skipping (${timeSince}s since last)`);
    return;
  }
  
  const watchlistEntries = Object.entries(state.watchlist || {});
  if (watchlistEntries.length < MIN_WATCHLIST_SIZE) {
    console.log(`⚠️ Watchlist too small (${watchlistEntries.length}), skipping cleanup`);
    return;
  }
  
  console.log(`\n🧹 ===== v45 CLEANUP START =====`);
  console.log(`📊 Before: ${watchlistEntries.length} tokens`);
  
  const removalReasons = { tooOld: 0, lowVolume: 0, flatPrice: 0, highRug: 0, lowHolders: 0, liquidityDrop: 0 };
  const tokensToRemove = [];
  
  for (const [address, token] of watchlistEntries) {
    if (state.armedTokens && state.armedTokens[address]) {
      continue; // Never remove armed tokens
    }
    
    let shouldRemove = false;
    let reason = "";
    
    if (isTokenTooOld(token)) {
      shouldRemove = true; reason = "too old"; removalReasons.tooOld++;
    } else if (hasLowVolume(token)) {
      shouldRemove = true; reason = "low volume"; removalReasons.lowVolume++;
    } else if (isPriceFlat(token)) {
      shouldRemove = true; reason = "flat price"; removalReasons.flatPrice++;
    } else if (hasHighRugRisk(token)) {
      shouldRemove = true; reason = "high rug"; removalReasons.highRug++;
    } else if (hasLowHolders(token)) {
      shouldRemove = true; reason = "low holders"; removalReasons.lowHolders++;
    } else if (hasLiquidityDrop(token)) {
      shouldRemove = true; reason = "liquidity drop"; removalReasons.liquidityDrop++;
    }
    
    if (shouldRemove) {
      tokensToRemove.push({ address, token, reason });
    }
  }
  
  console.log(`📋 Tokens to remove: ${tokensToRemove.length}`);
  console.log(`Reasons: Old=${removalReasons.tooOld}, Vol=${removalReasons.lowVolume}, Flat=${removalReasons.flatPrice}, Rug=${removalReasons.highRug}, Holders=${removalReasons.lowHolders}, Liq=${removalReasons.liquidityDrop}`);
  
  if (DRY_RUN) {
    console.log(`\n🧪 [DRY-RUN] Would remove ${tokensToRemove.length} tokens:`);
    tokensToRemove.slice(0, 10).forEach(item => {
      const age = ((Date.now() - item.token.discoveredAt) / 60000).toFixed(0);
      console.log(`  ❌ ${item.token.symbol || item.address.slice(0,8)}: ${item.reason} (age: ${age}m, vol: $${(item.token.volume24h || 0).toFixed(0)})`);
    });
    if (tokensToRemove.length > 10) console.log(`  ... and ${tokensToRemove.length - 10} more`);
    console.log(`🧪 [DRY-RUN] Set DRY_RUN=false to execute removal`);
  } else {
    for (const item of tokensToRemove) {
      delete state.watchlist[item.address];
      if (state.prices) delete state.prices[item.address];
    }
    console.log(`✅ Removed ${tokensToRemove.length} tokens`);
  }
  
  const remainingCount = Object.keys(state.watchlist || {}).length;
  console.log(`📊 After: ${remainingCount} tokens`);
  
  if (remainingCount > 150) {
    console.log(`\n⚠️ Still ${remainingCount} > 150. Using quality scoring...`);
    await keepTopTokens(state, 150);
  }
  
  lastCleanupTime = now;
  console.log(`🧹 ===== CLEANUP COMPLETE =====\n`);
}

// Keep Top N by Quality Score
async function keepTopTokens(state, maxTokens) {
  const tokens = Object.entries(state.watchlist || {});
  if (tokens.length <= maxTokens) return;
  
  console.log(`🎯 Scoring ${tokens.length} tokens...`);
  
  const scored = tokens.map(([addr, token]) => {
    const isArmed = state.armedTokens && state.armedTokens[addr];
    const score = isArmed ? 9999 : calculateQualityScore(token);
    return { address: addr, token, score, armed: isArmed };
  });
  
  scored.sort((a, b) => b.score - a.score);
  const toKeep = scored.slice(0, maxTokens);
  const toRemove = scored.slice(maxTokens);
  
  console.log(`📈 Top 5:`);
  toKeep.slice(0, 5).forEach((item, i) => {
    console.log(`${i+1}. ${item.token.symbol || item.address.slice(0,8)} (score: ${item.score})`);
  });
  
  if (DRY_RUN) {
    console.log(`🧪 [DRY-RUN] Would remove ${toRemove.length} low-score tokens`);
  } else {
    for (const item of toRemove) {
      delete state.watchlist[item.address];
      if (state.prices) delete state.prices[item.address];
    }
    console.log(`✅ Kept top ${toKeep.length} by quality`);
  }
}

console.log("✅ v45 Cleanup System Loaded (DRY_RUN=" + DRY_RUN + ")");
